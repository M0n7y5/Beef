# Leaks found at shutdown reach the Output panel, and a leaked object shows its allocation stack trace
SetExpectError("1 object memory leak detected")
RunWithCompiling()
ExpectError()

SelectCallStackWithStr("Internal.Shutdown")
AddWatch("(System.Object)(void*)LeakTest.Program.sLeakedAddr")
SelectWatch("(System.Object)(void*)LeakTest.Program.sLeakedAddr")
AssertSelectedWatchContains("{ 42 }")
OpenSelectedWatches()
SelectWatch("[AllocStackTrace]")
AssertSelectedWatchContains("LeakTest.Program.Leak()")
OpenSelectedWatches()
SelectWatch("[0]")
AssertSelectedWatchContains("LeakTest.Program.Leak()")

# The count is not checked: Windows reports one more raw block, untyped and without a stack trace
SetExpectError("raw memory leak")
Continue()
ExpectError()
Continue()
