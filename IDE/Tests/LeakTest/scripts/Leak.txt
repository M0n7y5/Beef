# Leaks found at shutdown reach the Output panel, and a leaked object shows its allocation stack trace
SetExpectError("1 object memory leak detected")
RunWithCompiling()
ExpectError()

# Needs a Beef frame, and the Win32 stack walk from the break in the debug runtime does not reach one
if (platform != "Win32") SelectCallStackWithStr("Internal.Shutdown")
if (platform != "Win32") AddWatch("(System.Object)(void*)LeakTest.Program.sLeakedAddr")
if (platform != "Win32") SelectWatch("(System.Object)(void*)LeakTest.Program.sLeakedAddr")
if (platform != "Win32") AssertSelectedWatchContains("{ 42 }")
if (platform != "Win32") OpenSelectedWatches()
if (platform != "Win32") SelectWatch("[AllocStackTrace]")
if (platform != "Win32") AssertSelectedWatchContains("LeakTest.Program.Leak()")
if (platform != "Win32") OpenSelectedWatches()
if (platform != "Win32") SelectWatch("[0]")
if (platform != "Win32") AssertSelectedWatchContains("LeakTest.Program.Leak()")

# The count is not checked: Windows reports one more raw block, untyped and without a stack trace
SetExpectError("raw memory leak")
Continue()
ExpectError()
Continue()
